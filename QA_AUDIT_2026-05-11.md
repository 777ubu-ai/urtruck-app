## D. SQL queries для тебя (выполнить руками на VPS)

⚠ **Перед любым DELETE — backup**. Эти запросы только SELECT.

```sql
-- ШАГ 1: backup перед любой работой
.backup /home/ubuntu/urtruck-security/database/security.db.bak.2026-05-11

-- ============================================
-- A. Counts по статусам
-- ============================================
SELECT 'cargos' tbl, status, COUNT(*) cnt FROM cargos GROUP BY status
UNION ALL SELECT 'trips', status, COUNT(*) FROM trips GROUP BY status
UNION ALL SELECT 'bids', status, COUNT(*) FROM bids GROUP BY status
UNION ALL SELECT 'deals', status, COUNT(*) FROM deals GROUP BY status;

-- ============================================
-- B. Тестовый мусор по bidder_id (НЕ удалять, только посчитать)
-- ============================================
SELECT 'guest_%' AS prefix, COUNT(*) AS n FROM bids WHERE bidder_id LIKE 'guest_%'
UNION ALL SELECT 'agent-%', COUNT(*) FROM bids WHERE bidder_id LIKE 'agent-%'
UNION ALL SELECT 'test%',   COUNT(*) FROM bids WHERE bidder_id LIKE 'test%'
UNION ALL SELECT 'mock%',   COUNT(*) FROM bids WHERE bidder_id LIKE 'mock%'
UNION ALL SELECT 'currency-regression%', COUNT(*) FROM bids WHERE bidder_id LIKE 'currency-regression%'
UNION ALL SELECT 'Direct probe%', COUNT(*) FROM bids WHERE bidder_id LIKE 'Direct probe%';

-- Те же префиксы в users / drivers
SELECT prefix, COUNT(*) FROM (
  SELECT CASE 
    WHEN id LIKE 'guest_%' THEN 'guest_'
    WHEN id LIKE 'agent-%' THEN 'agent-'
    WHEN id LIKE 'test%'   THEN 'test'
    WHEN id LIKE 'mock%'   THEN 'mock'
    ELSE 'real'
  END AS prefix
  FROM drivers_registration
) GROUP BY prefix;

-- ============================================
-- C. «Призраки» — карточки видны юзеру, но битые
-- ============================================
-- Активные грузы без from/to
SELECT id, owner_id, from_city, to_city, pickup_date, price, created_at
FROM cargos
WHERE status = 'active'
  AND (from_city IS NULL OR from_city = '' OR from_city = '—'
       OR to_city IS NULL OR to_city = '' OR to_city = '—');

-- Активные рейсы без from/to/departure
SELECT id, driver_id, from_city, to_city, departure
FROM trips
WHERE status = 'active'
  AND (from_city IS NULL OR from_city = '' OR to_city IS NULL OR to_city = '');

-- Активные грузы с pickup_date в прошлом
SELECT id, from_city, to_city, pickup_date
FROM cargos
WHERE status = 'active'
  AND pickup_date IS NOT NULL
  AND pickup_date != ''
  AND date(pickup_date) < date('now');

-- ============================================
-- D. bids_count не совпадает с реальным числом активных bids
-- ============================================
SELECT c.id, c.bids_count AS stored, 
  (SELECT COUNT(*) FROM bids b WHERE b.cargo_id = c.id AND b.status = 'pending') AS actual_pending,
  (SELECT COUNT(*) FROM bids b WHERE b.cargo_id = c.id AND b.status NOT IN ('cancelled','rejected')) AS actual_visible
FROM cargos c
WHERE c.status = 'active'
  AND c.bids_count != (SELECT COUNT(*) FROM bids b WHERE b.cargo_id = c.id AND b.status NOT IN ('cancelled','rejected'))
LIMIT 50;

-- ============================================
-- E. Notifications — типы и роли получателя
-- ============================================
SELECT type, recipient_role, COUNT(*) FROM notifications 
GROUP BY type, recipient_role 
ORDER BY 3 DESC;

-- ============================================
-- F. Users — отсутствует имя / город / роль
-- ============================================
SELECT 'no_name' AS issue, COUNT(*) FROM drivers_registration WHERE (full_name IS NULL OR full_name = '') AND status != 'guest'
UNION ALL SELECT 'no_city', COUNT(*) FROM drivers_registration WHERE (city IS NULL OR city = '') AND status != 'guest'
UNION ALL SELECT 'no_role', COUNT(*) FROM drivers_registration WHERE (role IS NULL OR role = '');

-- ============================================
-- G. Чат rooms без сообщений (orphan)
-- ============================================
SELECT r.id, r.created_at FROM chat_rooms r 
WHERE NOT EXISTS (SELECT 1 FROM chat_messages m WHERE m.room_id = r.id)
LIMIT 20;
```
